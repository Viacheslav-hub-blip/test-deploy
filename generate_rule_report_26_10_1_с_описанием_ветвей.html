WITH hist_features AS (
    SELECT
        event_id_src,
        user_id_src,
        COUNT(*) AS hist_n_events,
        COUNT(*) FILTER (WHERE event_type = 'CHANGE_PHONE') AS hist_n_change_phone,
        COUNT(*) FILTER (WHERE event_type = 'UPDATE_USER') AS hist_n_update_user,
        COUNT(*) FILTER (WHERE event_type = 'PAYMENT') AS hist_n_payment,
        COUNT(*) FILTER (WHERE event_type = 'CHECK') AS hist_n_check,
        COUNT(*) FILTER (WHERE sub_channel = 'ISSUER') AS hist_n_issuer_events,
        COUNT(DISTINCT event_dt) AS hist_n_unique_dates,
        COUNT(*) FILTER (WHERE event_channel = 'BRANCH') AS hist_branch_events,
        COUNT(*) FILTER (WHERE event_channel = 'CALL_CENTER') AS hist_call_center_events,
        COUNT(*) FILTER (WHERE sub_type = 'MB_ON') AS hist_mb_on,
        AVG(CAST(transaction_amount AS FLOAT64)) AS hist_mean_amount,
        MAX(CAST(transaction_amount AS FLOAT64)) AS hist_max_amount,
        SUM(CAST(transaction_amount AS FLOAT64)) AS hist_sum_amount
    FROM hist_table
    WHERE is_src_event = false
    GROUP BY event_id_src, user_id_src
),

ops_with_features AS (
    SELECT
        o.*,
        COALESCE(hf.hist_n_events, 0) AS hist_n_events,
        COALESCE(hf.hist_n_change_phone, 0) AS hist_n_change_phone,
        COALESCE(hf.hist_n_update_user, 0) AS hist_n_update_user,
        COALESCE(hf.hist_n_payment, 0) AS hist_n_payment,
        COALESCE(hf.hist_n_check, 0) AS hist_n_check,
        COALESCE(hf.hist_n_issuer_events, 0) AS hist_n_issuer_events,
        COALESCE(hf.hist_n_unique_dates, 0) AS hist_n_unique_dates,
        COALESCE(hf.hist_branch_events, 0) AS hist_branch_events,
        COALESCE(hf.hist_call_center_events, 0) AS hist_call_center_events,
        COALESCE(hf.hist_mb_on, 0) AS hist_mb_on,
        COALESCE(hf.hist_mean_amount, 0) AS hist_mean_amount,
        COALESCE(hf.hist_max_amount, 0) AS hist_max_amount,
        COALESCE(hf.hist_sum_amount, 0) AS hist_sum_amount
    FROM ops_table o
    LEFT JOIN hist_features hf
        ON o.event_id = hf.event_id_src
       AND o.user_id = hf.user_id_src
)
SELECT * FROM ops_with_features;
